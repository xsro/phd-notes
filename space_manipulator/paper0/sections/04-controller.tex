\section{控制算法设计}
\label{sec:controller}

本节详细设计"扰动观测--动力学补偿--误差线性化--周期延迟反馈"控制结构的各组成部分。第~\ref{sec:ptdo}~节构造预设时间扰动观测器以估计集总扰动；第~\ref{sec:compensation}~节利用计算力矩补偿将误差动力学线性化；第~\ref{sec:pdf_design}~节在可控线性误差通道中嵌入光滑周期延迟反馈，给出满足单周期零化条件的 PDF 增益构造。

\subsection{预设时间扰动观测器}
\label{sec:ptdo}

对等效关节空间模型~\eqref{eq:joint_model_relabelled}，定义关节速度为
\[
\chi = \dot q_m \in\mathbb R^n.
\]
将式~\eqref{eq:joint_model_relabelled}~改写为加速度通道形式
\begin{equation}
\dot\chi = u_o + \Delta_a,\qquad
u_o = M_e^{-1}(q_m)\bigl[\tau - C_e(q_m,\dot q_m)\dot q_m\bigr],\qquad
\Delta_a = M_e^{-1}(q_m)d(t).
\label{eq:observer_channel}
\end{equation}
其中 \(u_o\) 为由控制输入和速度相关项构成的可计算加速度，\(\Delta_a\) 为匹配到加速度通道的集总扰动。该形式与预设时间扰动观测器的标准受扰单积分器模型一致~\cite{jiangPrescribedtimeDisturbanceObserver2024}。

\noindent 令 \(z_1,z_2\in\mathbb R^n\) 分别为 \(\chi\) 和 \(\Delta_a\) 的估计值，定义可测复合误差
\begin{equation}
\varepsilon_1 = \chi - z_1 - \bar\xi(t),
\label{eq:observer_error}
\end{equation}
其中 \(\bar\xi(t)\in\mathbb R^n\) 为调节函数，满足以下条件：

\begin{enumerate}
  \item \(\bar\xi(t)\) 光滑有界，单调衰减；
  \item \(\bar\xi(t)=0,\quad \forall\,t\ge T_o\)。
\end{enumerate}

例如可取如下二次衰减形式
\[
\bar\xi(t) = 
\begin{cases}
\bar\xi_0\,(T_o-t)^2, & 0\le t<T_o,\\
0, & t\ge T_o,
\end{cases}
\]
其中 \(\bar\xi_0\in\mathbb R^n\) 为常数向量。定义 \(\operatorname{sig}^{\alpha}(x)=\operatorname{sgn}(x)|x|^\alpha\)，并按元素作用于向量。

\begin{theorem}[预设时间扰动观测器]\label{thm:ptdo}
  考虑系统~\eqref{eq:observer_channel}。在假设~\ref{as:disturbance}~下，构造如下预设时间扰动观测器
  \begin{equation}
  \left\{
  \begin{aligned}
  \dot z_1 &= z_2 + \frac{\pi}{\eta T_c}
             \phi_1\!\left(\frac{\varepsilon_1}{\sigma}\right)
             - \dot{\bar\xi}(t) + \bar\xi(t) + u_o, \\[4pt]
  \dot z_2 &= \frac{\pi}{\sigma\eta T_c}
             \phi_2\!\left(\frac{\varepsilon_1}{\sigma}\right)
             - \dot{\bar\xi}(t),
  \end{aligned}
  \right.
  \label{eq:ptdo}
  \end{equation}
  其中 \(T_c\le T_o\) 为可设计收敛时间参数，\(\eta\in(0,1)\)，\(\sigma>0\)，且
  \begin{align}
  \phi_1(x) &= \operatorname{sig}^{1-\eta/2}(x) + \operatorname{sig}^{1+\eta/2}(x), \label{eq:phi1}\\
  \phi_2(x) &= \operatorname{sig}^{2-\eta}(x) + \operatorname{sig}^{2+\eta}(x) + \sigma\operatorname{sgn}(x). \label{eq:phi2}
  \end{align}
  则在连续时间实现条件下，观测误差满足
  \begin{equation}
  \hat\Delta_a(t) = \Delta_a(t),\qquad
  \hat d(t) = d(t),\qquad \forall\,t\ge T_o,
  \label{eq:observer_convergence}
  \end{equation}
  其中 \(\hat\Delta_a = z_2\)，\(\hat d(t) = M_e(q_m)\hat\Delta_a(t)\)。
\end{theorem}

\begin{proof}
  定义观测误差 \(\varepsilon_2 = \Delta_a - z_2\)。由式~\eqref{eq:observer_channel}~和式~\eqref{eq:ptdo}~可得误差动态
  \begin{align*}
  \dot\varepsilon_1 &= -\frac{\pi}{\eta T_c}\phi_1(\varepsilon_1/\sigma) + \varepsilon_2,\\
  \dot\varepsilon_2 &= \dot\Delta_a - \frac{\pi}{\sigma\eta T_c}\phi_2(\varepsilon_1/\sigma) + \dot{\bar\xi}(t).
  \end{align*}
  与文献~\cite{jiangPrescribedtimeDisturbanceObserver2024}~的结构一致，在 \(\Delta_a\) 和 \(\dot\Delta_a\) 有界的条件下，可通过调节函数 \(\bar\xi(t)\) 的衰减性质和非线性函数 \(\phi_1,\phi_2\) 的齐次性保证 \(\varepsilon_1(t),\varepsilon_2(t)\to 0\) 在 \(t\to T_o\) 时成立。收敛时间由参数 \(T_c\) 和 \(\eta\) 共同控制，且可通过适当选取 \(T_c\le T_o\) 使收敛在设计时间 \(T_o\) 前完成。由 \(\hat d = M_e\hat\Delta_a\) 立即得到力矩级扰动的收敛性。
\end{proof}

\begin{remark}[观测器参数的实用选取]
  参数 \(T_o\) 和 \(T_c\) 不宜选得过小，否则可能带来显著的观测峰值，进而通过扰动补偿通道引发过大的控制输入峰值，影响执行器的实际可行性。
\end{remark}

\subsection{动力学补偿与误差线性化}
\label{sec:compensation}

在获得扰动估计 \(\hat d(t)\) 后，采用如下计算力矩补偿结构
\begin{equation}
\tau = M_e(q_m)v + C_e(q_m,\dot q_m)\dot q_m - \hat d(t),
\label{eq:computed_torque}
\end{equation}
其中 \(v\in\mathbb R^n\) 为待设计的辅助加速度输入。

\begin{lemma}[误差通道线性化]\label{lem:linearization}
  令 \(v = \ddot q_d + \nu\)，其中 \(\nu\in\mathbb R^n\) 为辅助控制输入。假设以下两条件同时成立：
  \begin{enumerate}
    \item[(C1)] 扰动观测器已收敛，即 \(\hat d(t) = d(t)\)；
    \item[(C2)] 等效惯性矩阵精确已知，即 \(M_e(q_m)\) 真值可获取。
  \end{enumerate}
  则将式~\eqref{eq:computed_torque}~代入式~\eqref{eq:joint_model_relabelled}~可得
  \begin{equation}
  \ddot q_m = v.
  \label{eq:double_integrator}
  \end{equation}
  进而，定义跟踪误差 \(e = q_m - q_d\)，误差状态 \(z = [e^\top,\ \dot e^\top]^\top\)，则有
  \begin{equation}
  \dot z = A z + B\nu,
  \label{eq:linear_error}
  \end{equation}
  其中
  \begin{equation}
  A = \begin{bmatrix} 0 & I_n \\ 0 & 0 \end{bmatrix} \in\mathbb R^{2n\times2n},\qquad
  B = \begin{bmatrix} 0 \\ I_n \end{bmatrix} \in\mathbb R^{2n\times n}.
  \label{eq:ab_matrices}
  \end{equation}
  矩阵对 \((A,B)\) 可控。
\end{lemma}

\begin{proof}
  在条件 (C1) 和 (C2) 下，将式~\eqref{eq:computed_torque}~代入式~\eqref{eq:joint_model_relabelled}~得
  \[
  M_e\ddot q_m + C_e\dot q_m = M_e v + C_e\dot q_m - \hat d + d = M_e v.
  \]
  由假设~\ref{as:inertia}，\(M_e\) 一致正定，左乘 \(M_e^{-1}\) 即得 \(\ddot q_m = v\)。再将 \(v = \ddot q_d + \nu\) 代入，有
  \[
  \ddot e = \ddot q_m - \ddot q_d = \nu.
  \]
  由此直接得到式~\eqref{eq:linear_error}。可控性由 Kalman 秩条件验证：
  \[
  \operatorname{rank}\bigl[B,\ AB\bigr] 
  = \operatorname{rank}\begin{bmatrix}0 & I_n \\ I_n & 0\end{bmatrix}
  = 2n.
  \]
\end{proof}

引理~\ref{lem:linearization}~是本文方法成立的关键桥梁：它将自由漂浮空间机械臂的非线性跟踪控制问题转化为线性可控系统的镇定问题。PDF 理论只需作用在式~\eqref{eq:linear_error}~上，而不直接处理原系统的非线性耦合项。

\subsection{PDF 线性误差系统设计}
\label{sec:pdf_design}

在获得线性误差模型~\eqref{eq:linear_error}~后，本节引入光滑周期延迟反馈实现预设时间镇定。

\subsubsection{基准闭环与周期延迟结构}

首先选取常值反馈增益 \(K_0\in\mathbb R^{n\times2n}\)，使
\begin{equation}
A_c = A + BK_0
\label{eq:ac_matrix}
\end{equation}
成为期望的 Hurwitz 基准闭环矩阵。由于 \((A,B)\) 可控，\(K_0\) 的存在性由极点配置保证。在此基础上，引入周期延迟反馈
\begin{equation}
\nu(t) = K_0 z(t) - K_c(t)z(t-h),
\label{eq:pdf_error_input}
\end{equation}
其中 \(K_c(t)\in\mathbb R^{n\times2n}\) 为 \(2h\) 周期时变增益，满足 \(K_c(t)=0,\ \forall\,t\in[0,h)\)。将式~\eqref{eq:pdf_error_input}~代入式~\eqref{eq:linear_error}，得到延迟闭环系统
\begin{equation}
\dot z(t) = A_c z(t) - BK_c(t)z(t-h).
\label{eq:closed_delay_error}
\end{equation}

\subsubsection{Gramian 型 PDF 增益构造}

令 \(R_h(t)\) 为满足以下条件的标量函数：
\begin{itemize}
  \item \(R_h(t+2h)=R_h(t)\)（周期 \(2h\)）；
  \item \(R_h(t)=0,\quad \forall\,t\in[0,h]\)；
  \item \(R_h(t)\ge 0,\quad \forall\,t\in[h,2h]\)，且在 \([h,2h]\) 的某个具有非零测度的子区间上严格为正；
  \item \(R_h(t)\) 在定义域上光滑（连续可微至所需阶数）。
\end{itemize}

\begin{definition}[可控 Gramian]\label{def:gramian}
  对给定的 \(A_c\in\mathbb R^{2n\times2n}\)、\(B\in\mathbb R^{2n\times n}\)、时滞 \(h>0\) 和周期函数 \(R_h(t)\)，定义矩阵
  \begin{equation}
  W_c(A_c,h) = \int_h^{2h} e^{-A_c s} B R_h(s) B^\top e^{-A_c^\top s}\,ds.
  \label{eq:wc_matrix}
  \end{equation}
  由于 \((A_c,B)\) 可控且 \(R_h(t)\) 在后半周期的非零测度子区间上严格为正，\(W_c(A_c,h)\) 为正定矩阵。
\end{definition}

\begin{theorem}[PDF 单周期零化增益]\label{thm:pdf_gain}
  对于延迟闭环系统~\eqref{eq:closed_delay_error}，定义周期时变增益为
  \begin{equation}
  K_c(t) = 
  \begin{cases}
  0, & 0\le t < h,\\[4pt]
  R_h(t)B^\top e^{-A_c^\top t} W_c^{-1}(A_c,h) e^{A_c(h-t)}, & h\le t < 2h,
  \end{cases}
  \label{eq:kc_pdf}
  \end{equation}
  并以周期 \(2h\) 延拓。则该闭环系统的单周期状态转移矩阵满足
  \begin{equation}
  \Delta_K(h) = 0,
  \label{eq:single_period_null}
  \end{equation}
  即实现了定义~\ref{def:single_period_nullification}~所述的单周期零化。因此，对任意初始条件，系统~\eqref{eq:closed_delay_error}~的解满足
  \begin{equation}
  z(t) = 0,\qquad \forall\,t\ge 2h.
  \label{eq:pdf_convergence}
  \end{equation}
\end{theorem}

\begin{proof}
  由分段结构和分步法，将 \([0,2h]\) 分为两段分析。在前半段 \([0,h)\) 上，\(K_c(t)=0\)，系统退化为无延迟的线性系统 \(\dot z = A_c z\)，其解为 \(z(t)=e^{A_c t}z(0)\)。特别地，在 \(t=h\) 处有 \(z(h)=e^{A_c h}z(0)\)。

  在后半段 \([h,2h]\) 上，系统为时滞系统 \(\dot z(t)=A_c z(t)-BK_c(t)z(t-h)\)。对任意 \(t\in[h,2h]\)，利用变分公式，引入 \(t-h\in[0,h]\)，因此 \(z(t-h)=e^{A_c(t-h)}z(0)\)。将改写后的方程从 \(s=h\) 积分至 \(s=t\)：
  \[
  z(t) = e^{A_c(t-h)}z(h) - \int_h^t e^{A_c(t-s)} B K_c(s)\, e^{A_c(s-h)}z(0)\,ds.
  \]
  代入 \(z(h)=e^{A_c h}z(0)\)，有
  \[
  z(t) = e^{A_c t}z(0) - \int_h^t e^{A_c(t-s)} B K_c(s)\, e^{A_c(s-h)}\,ds\,z(0).
  \]
  在 \(t=2h\) 处，
  \[
  z(2h) = e^{A_c(2h)}z(0) - \int_h^{2h} e^{A_c(2h-s)} B K_c(s)\, e^{A_c(s-h)}ds\,z(0).
  \]
  将式~\eqref{eq:kc_pdf}~代入积分项，可得
  \[
  \int_h^{2h} e^{A_c(2h-s)} B K_c(s) e^{A_c(s-h)}ds 
  = e^{A_c h}\int_h^{2h} e^{-A_c s} B R_h(s) B^\top e^{-A_c^\top s} ds\, W_c^{-1}(A_c,h)\, e^{A_c h}.
  \]
  由定义~\ref{def:gramian}，右端积分即为 \(W_c(A_c,h)\)，于是
  \[
  \int_h^{2h} e^{A_c(2h-s)} B K_c(s) e^{A_c(s-h)}ds = e^{A_c h} I_{2n} e^{A_c h} = e^{A_c(2h)}.
  \]
  代入 \(z(2h)\) 表达式得
  \[
  z(2h) = e^{A_c(2h)}z(0) - e^{A_c(2h)}z(0) = 0.
  \]
  因此 \(\Delta_K(h)=0\)。重复应用该映射知，\(z(t)=0,\ \forall t\ge 2h\)。
\end{proof}

\begin{remark}[增益的离线计算]
  式~\eqref{eq:kc_pdf}~中的 \(W_c(A_c,h)\) 和 \(K_c(t)\) 可在控制运行前离线积分计算。在线控制时仅需在每一采样时刻根据本地时间 \(\theta=t\bmod 2h\) 查表读取对应的增益值，并与当前误差 \(z(t)\) 及历史误差 \(z(t-h)\) 进行乘法运算。这一结构使得在线计算量与常规 PD 控制器相当。
\end{remark}

\subsubsection{完整控制律}

综合式~\eqref{eq:computed_torque}、\eqref{eq:pdf_error_input}，PDF 关节力矩控制律为
\begin{equation}
\tau(t) = M_e(q_m)\Bigl[\ddot q_d + K_0 z(t) - K_c(t)z(t-h)\Bigr] + C_e(q_m,\dot q_m)\dot q_m - \hat d(t).
\label{eq:final_controller}
\end{equation}

\subsection{两阶段时序设计}

在观测器和 PDF 控制器同时从 \(t=0\) 启动的情况下，\(0\le t<T_o\) 时间区间内的观测误差会作为残余扰动进入误差系统，使定理~\ref{thm:pdf_gain}~的理想单周期零化结论在该时段内不成立。为保持理论分析的清晰性，本文采用以下两阶段时序：

\begin{enumerate}
  \item \textbf{观测阶段：} \(t\in[0,T_o)\)，扰动观测器式~\eqref{eq:ptdo}~运行并收敛；PDF 反馈增益设为 \(K_c(t)=0\)，即仅由 \(K_0\) 提供基准反馈。
  \item \textbf{PDF 阶段：} \(t\in[T_o,\infty)\)，扰动观测器保持收敛状态 \(\hat d(t)=d(t)\)；PDF 控制器以局部时间 \(\vartheta = t-T_o\) 激活完整的周期延迟反馈增益 \(K_c(\vartheta)\)。
\end{enumerate}

在上述两阶段时序下，定理~\ref{thm:ptdo}~和定理~\ref{thm:pdf_gain}~的结论可分别应用于各自的时段。联合后的总预设时间上界为
\begin{equation}
T_{\mathrm{pre}} = T_o + 2h.
\label{eq:prescribed_total_time}
\end{equation}
式~\eqref{eq:prescribed_total_time}~中的 \(T_o\) 和 \(h\) 均为控制器设计阶段选定的参数，与初始误差大小无关，体现了"预设时间"的含义。
