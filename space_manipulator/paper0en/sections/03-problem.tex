\section{Problem Statement}\label{sec:problem}
Given a smooth bounded reference $q_d(t),\dot q_d(t),\ddot q_d(t)$, define
\begin{equation}e=q_m-q_d,\qquad\dot e=\dot q_m-\dot q_d,\qquad z=[e^\top,\dot e^\top]^\top.\label{eq:tracking_errors}\end{equation}
\begin{problem}[Prescribed-time joint trajectory tracking]\label{prob:tracking}
Design $\tau(t)$ such that, for a prescribed $T>0$, $e(t)=\dot e(t)=0$ for all $t\ge T$ under ideal continuous-time exact compensation. With nonidealities, require $\|z(t)\|\le\delta(\bar\Delta_r)$, where $\delta$ is increasing and $\delta(0)=0$.
\end{problem}
\subsection{Basic Assumptions}
\begin{assumption}[Invertible equivalent inertia]\label{as:inertia} $M_e(q_m)\succeq\underline m I_n$ and its nominal value is available.\end{assumption}
\begin{assumption}[Bounded reference]\label{as:reference} $q_d,\dot q_d,\ddot q_d$ are bounded and continuously differentiable; $q_m,\dot q_m$ are measured.\end{assumption}
\begin{assumption}[Available delay history]\label{as:delay_history} The history $z(t-h)$ is available from a buffer, with bounded piecewise-continuous initial function.\end{assumption}
\begin{assumption}[Bounded disturbance]\label{as:disturbance} $\Delta_a=M_e^{-1}d$ and $\dot\Delta_a$ are uniformly bounded.\end{assumption}
\begin{assumption}[Ideal conditions]\label{as:ideal} Exact prescribed-time tracking assumes continuous time, no saturation, exact $M_e,C_e$, and exact observer convergence; otherwise only practical bounded tracking is claimed.\end{assumption}
\subsection{Preliminaries: Smooth Periodic Delayed Feedback}
Consider $\dot x(t)=F(t)x(t)+G(t)x(t-h)$, with $2h$-periodic coefficients and $G(t)=0$ on the first half-period. Then
\begin{equation}x\bigl(2(k+1)h\bigr)=\Delta(h)x(2kh),\qquad k=0,1,2,\ldots.\label{eq:period_map_abstract}\end{equation}
The one-period matrix has the explicit representation
\begin{equation}\Delta(h)=\Phi(2h,0)+\int_h^{2h}\Phi(2h,s)G_0(s)\Phi(s-h,0)\,ds.\label{eq:period_map_explicit}\end{equation}
\begin{definition}[Single-period nullification]\label{def:single_period_nullification}
If $\Delta(h)=0$, then $x(t)=0$ for all $t\ge2h$.
\end{definition}
\begin{definition}[Nilpotent nullification]\label{def:nilpotent_nullification}
If $\Delta(h)$ is nilpotent of index $\nu_0$, then $x(t)=0$ for all $t\ge2\nu_0h$.
\end{definition}
