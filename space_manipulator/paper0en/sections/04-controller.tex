\section{Controller Design}\label{sec:controller}

This section separates the observer result from the delayed-feedback result. The
latter is proved exactly; the observer is required to satisfy the explicit
finite-time interface stated in Lemma~\ref{lem:observer_interface}.

\subsection{Disturbance-observer interface}\label{sec:ptdo}
Let $\chi=\dot q_m$ and define
\begin{equation}
\begin{aligned}
 \dot\chi&=u_o+\Delta_a,\\
 u_o&=M_e^{-1}(q_m)[\tau-C_e(q_m,\dot q_m)\dot q_m],\\
 \Delta_a&=M_e^{-1}(q_m)d.
\end{aligned}
 \label{eq:observer_channel}
\end{equation}
The following assumption is the precise interface needed by the controller.
\begin{assumption}[Observer interface]\label{as:observer_interface}
There are $T_o>0$ and an estimate $\hat\Delta_a$ such that, under the
continuous-time observer, $\hat\Delta_a(t)=\Delta_a(t)$ for all $t\geq T_o$.
Before $T_o$, the observation error is bounded. The estimate is formed as
$\hat d=M_e\hat\Delta_a$.
\end{assumption}

For completeness, the observer used in the simulations is the prescribed-time
homogeneous observer
\begin{equation}
\begin{aligned}
 \dot z_1&=z_2+u_o+\ell_1(t)\,\phi_1(\varepsilon_1/\sigma),\\
 \dot z_2&=\ell_2(t)\,\phi_2(\varepsilon_1/\sigma),
 \qquad \varepsilon_1=\chi-z_1,
\end{aligned}\label{eq:ptdo}
\end{equation}
where
\begin{align}
 \phi_1(x)&=\operatorname{sig}^{1-\eta/2}(x)+
             \operatorname{sig}^{1+\eta/2}(x),\\
 \phi_2(x)&=\operatorname{sig}^{1-\eta}(x)+
             \operatorname{sig}^{1+\eta}(x),
 \end{align}
$0<\eta<1$, $\operatorname{sig}^{\alpha}(x)=|x|^{\alpha}\operatorname{sgn}(x)$, and
the gains are prescribed-time functions of the designable observation time $T_o$,
\begin{equation}
 \ell_1(t)=\frac{m_1}{T_o-t},\qquad
 \ell_2(t)=\frac{m_2}{(T_o-t)^2},\qquad t<T_o,
 \label{eq:ptdo_gains}
\end{equation}
with $m_1,m_2>0$. The pair of powers in \eqref{eq:ptdo} is consistent with the
dilation weights $(1,1-\eta/2)$ and $(1,1+\eta/2)$ of the two homogeneous
branches, so that the error system of the observer possesses a negative
homogeneity degree in the stretched time
$s=-\ln(1-t/T_o)$ and therefore converges in finite stretched time, i.e., at
$t=T_o$ in the original time. The design follows the prescribed-time
homogeneous-observer construction of~\cite{jiangPrescribedtimeDisturbanceObserver2024};
the gains must be selected using the disturbance regularity bounds required by
that design, so that Assumption~\ref{as:disturbance} alone is not sufficient to
establish exact convergence of~\eqref{eq:ptdo}.
\begin{remark}[Gain regularisation in the implementation]
For $t\ge T_o-\delta$ the gains \eqref{eq:ptdo_gains} are frozen at their values
at $T_o-\delta$. The frozen observer is a high-gain second-order differentiator
whose residual estimation error is $O(\|\dot\Delta_a\|\delta^2/m_2)$; this
residual is part of $\Delta_r$ in Section~\ref{sec:practical} and does not alter
the stability statement, only the size of the practical bound.
\end{remark}

\begin{lemma}[Observer interface lemma]\label{lem:observer_interface}
Suppose the hypotheses of the prescribed-time homogeneous-observer theorem
used to select $m_1,m_2,\eta,\sigma$ hold, including the required bound on
$\dot\Delta_a$. Then the Filippov solution of~\eqref{eq:ptdo} satisfies
$z_2(t)=\Delta_a(t)$ for all $t\geq T_o$.
\end{lemma}
\begin{proof}
Subtracting~\eqref{eq:ptdo} from~\eqref{eq:observer_channel} gives the error
system
\begin{equation}
 \dot\varepsilon_1=\varepsilon_2-\ell_1(t)\phi_1(\varepsilon_1/\sigma),\qquad
 \dot\varepsilon_2=-\ell_2(t)\phi_2(\varepsilon_1/\sigma)+\dot\Delta_a,
 \label{eq:observer_error}
\end{equation}
with $\varepsilon_1=\chi-z_1$ and $\varepsilon_2=\Delta_a-z_2$. Because
$\phi_1$ and $\phi_2$ are odd, continuous and strictly increasing, and because
the powers in~\eqref{eq:observer_error} satisfy the dilation-consistency
condition $1-\eta=2(1-\eta/2)-1$ (and likewise for the upper branch), the
unperturbed vector field is homogeneous with weights $(1,1-\eta/2)$ and
$(1,1+\eta/2)$, of negative homogeneity degree on both branches. Under the
stretched time $s=-\ln(1-t/T_o)$, the time-varying gains
\eqref{eq:ptdo_gains} become constant, so the unperturbed system is finite-time
stable in $s$ and the cited theorem provides a Lyapunov function $V$ and
constants $c_1,c_2>0$ with $\dot V\leq-c_1V^{\alpha}-c_2V^{\beta}$,
$0<\alpha<1<\beta$, in the stretched time. The standard fixed-time integral
bound then gives convergence in finite $s$, that is, at $t=T_o$ in the original
time. Since $\Delta_a$ is absolutely continuous with bounded derivative
(Assumption~\ref{as:disturbance}), the perturbation $\dot\Delta_a$ is bounded,
and homogeneity plus the vanishing of the injection as $\varepsilon_1\to0$
ensure that the perturbed solution enters every neighbourhood of the origin
before $T_o$. Strong invariance of the set
$\{(\varepsilon_1,\varepsilon_2):\varepsilon_1=\varepsilon_2=0\}$ for the
continuous-time observer then gives $z_2(t)=\Delta_a(t)$ for all $t\geq T_o$.
\end{proof}
\begin{remark}
Lemma~\ref{lem:observer_interface} is stated for the ideal homogeneous observer.
The regularised implementation of the previous remark replaces exact
convergence by the residual $O(\|\dot\Delta_a\|\delta^2/m_2)$ that appears in
$\Delta_r$ of Section~\ref{sec:practical}; the simulation results of
Section~\ref{sec:simulation} show that this residual is two orders of magnitude
smaller than the disturbance it compensates.
\end{remark}

\subsection{Dynamic compensation and error linearization}
Use
\begin{equation}
 \tau=M_e(q_m)v+C_e(q_m,\dot q_m)\dot q_m-\hat d,
 \qquad v=\ddot q_d+\nu.
 \label{eq:computed_torque}
\end{equation}
Under exact model matching and Assumption~\ref{as:observer_interface},
$\hat d=d$ for $t\geq T_o$, and~\eqref{eq:joint_model_relabelled} gives
$\ddot q_m=v$. Consequently, with
$e=q_m-q_d$ and $z=[e^\top,\dot e^\top]^\top$,
\begin{equation}
 \dot z=Az+B\nu,
 \qquad A=\begin{bmatrix}0&I_n\\0&0\end{bmatrix},
 \qquad B=\begin{bmatrix}0\\I_n\end{bmatrix}.
 \label{eq:linear_error}
\end{equation}
The pair $(A,B)$ is controllable since
$\operatorname{rank}[B,AB]=2n$.

\subsection{Smooth periodic delayed feedback}\label{sec:pdf_design}
Choose $K_0$ so that $A_c=A+BK_0$ is Hurwitz. Let $h>0$ and let
$R_h:[0,2h]\to\mathbb R$ be smooth, nonnegative, zero on $[0,h]$, and
positive on a subset of positive measure in $(h,2h)$. Define
\begin{equation}
 W_c=\int_h^{2h}e^{-A_cs}B R_h(s)B^\top e^{-A_c^\top s}\,ds.
 \label{eq:wc_matrix}
\end{equation}
\begin{lemma}[Positivity of the PDF Gramian]\label{lem:gramian}
If $(A_c,B)$ is controllable, then $W_c\succ0$.
\end{lemma}
\begin{proof}
For any $x\in\mathbb R^{2n}$,
\[
 x^\top W_cx=\int_h^{2h}R_h(s)\|B^\top e^{-A_c^\top s}x\|^2ds\geq0.
\]
If this quantity is zero, continuity implies
$B^\top e^{-A_c^\top s}x=0$ on an interval contained in the set where
$R_h>0$. The left-hand side is analytic in $s$, hence it vanishes for all
$s$. Differentiating at any point gives
$B^\top(A_c^\top)^kx=0$ for every $k\geq0$. Controllability of
$(A_c,B)$ implies $x=0$. Therefore $W_c$ is positive definite.
\end{proof}

For local PDF time $\vartheta=t-T_o$, define the $2h$-periodic gain
\begin{equation}
 K_c(\vartheta)=\begin{cases}
 0,&0\leq\vartheta<h,\\
 R_h(\vartheta)B^\top e^{-A_c^\top\vartheta}W_c^{-1}
 e^{A_c(h-\vartheta)},&h\leq\vartheta<2h,
 \end{cases}
 \label{eq:kc_pdf}
\end{equation}
and extend it periodically. The control input is
\begin{equation}
 \nu(t)=K_0z(t)-K_c(t-T_o)z(t-h),\qquad t\geq T_o.
 \label{eq:pdf_error_input}
\end{equation}
Thus the complete controller is
\begin{equation}
 \tau=M_e\left[\ddot q_d+K_0z-K_c(t-T_o)z(t-h)\right]
 +C_e\dot q_m-\hat d.
 \label{eq:final_controller}
\end{equation}
