\section{Stability and Convergence Analysis}\label{sec:stability}

\subsection{Exact prescribed-time tracking}
Define the local PDF time $\vartheta=t-T_o$. Before the observer settling
instant, the observer error is bounded by Assumption~\ref{as:observer_interface}.
For $t\in[T_o,T_o+h)$, $K_c(\vartheta)=0$ and the compensated error dynamics
are $\dot z=A_cz$. Hence
\begin{equation}
 z(T_o+h)=e^{A_ch}z(T_o).
 \label{eq:phaseI_bound}
\end{equation}
In particular, the state entering the second half-period is finite.

\begin{theorem}[PDF one-period nullification]\label{thm:pdf_nullification}
Under Assumptions~\ref{as:inertia}--\ref{as:ideal}, the observer interface,
and $W_c\succ0$, the ideal homogeneous error system under
~\eqref{eq:pdf_error_input} satisfies
\begin{equation}
 z(T_o+2h)=0
\end{equation}
for every finite $z(T_o)$. Moreover, $z(t)=0$ for every $t\geq T_o+2h$.
\end{theorem}
\begin{proof}
For $0\leq\vartheta<h$, the delayed term is multiplied by zero, so
$z(T_o+\vartheta)=e^{A_c\vartheta}z(T_o)$. In particular,
$z(T_o+h)=e^{A_ch}z(T_o)$. For $h\leq\vartheta\leq2h$, variation of
constants gives
\begin{align}
 z(T_o+2h)
 &=e^{2A_ch}z(T_o)\\
 &\quad-\int_h^{2h}e^{A_c(2h-s)}B K_c(s)e^{A_c(s-h)}ds\,z(T_o).
 \label{eq:pdf_variation_constants}
\end{align}
Using~\eqref{eq:kc_pdf}, the integral term equals
\begin{align*}
 &\int_h^{2h}e^{A_c(2h-s)}BR_h(s)B^\top e^{-A_c^\top s}
 W_c^{-1}e^{A_c(h-s)}e^{A_c(s-h)}ds\\
 &=e^{2A_ch}\left[\int_h^{2h}e^{-A_cs}BR_h(s)B^\top
 e^{-A_c^\top s}ds\right]W_c^{-1}\\
 &=e^{2A_ch}W_cW_c^{-1}=e^{2A_ch}.
\end{align*}
Therefore $z(T_o+2h)=0$. Since the delayed history is zero after this
instant and the system is homogeneous, uniqueness implies $z(t)=0$ for all
later time.\end{proof}

\begin{theorem}[Exact prescribed-time tracking]\label{thm:main_ideal}
Under the assumptions of Theorem~\ref{thm:pdf_nullification},
\begin{equation}
 e(t)=0,\qquad \dot e(t)=0,
 \qquad t\geq T_o+2h=T.
\end{equation}
The settling time is independent of the initial tracking error.
\end{theorem}
\begin{proof}
The observer interface gives exact disturbance compensation after $T_o$.
The computed-torque calculation in Section~\ref{sec:controller} therefore
reduces the tracking error to the PDF system. Theorem~\ref{thm:pdf_nullification}
then gives $z(t)=0$ after $T_o+2h$.\end{proof}

\subsection{Practical bounded tracking}\label{sec:practical}
Let the implemented model and observer generate a residual acceleration
term $\Delta_r$. The perturbed error system is
\begin{equation}
 \dot z=A_cz-BK_c(\vartheta)z(t-h)+B\Delta_r(t),
 \qquad \vartheta=t-T_o.
 \label{eq:perturbed_delay_error}
\end{equation}
Assume
\begin{equation}
 \|\Delta_r(t)\|\leq\bar\Delta_r,
 \qquad t\geq T_o.
 \label{eq:residual_bound}
\end{equation}
Let $\Psi(\vartheta,s)$ be the causal transition operator of the homogeneous
PDF delay system. Define the finite constants
\begin{align}
 \Gamma_h&:=\sup_{0\leq\vartheta\leq2h}
 \int_0^\vartheta\|\Psi(\vartheta,s)B\|\,ds,\\
 \Gamma_h^*&:=\sup_{0\leq\vartheta\leq2h}
 \int_0^\vartheta\|\Psi(\vartheta,s)B\|\,ds.
 \label{eq:gamma_definition}
\end{align}
For the periodically nullified system, the same finite within-period gain can
be used after every period boundary: the homogeneous period-boundary state is
zero and the within-period transition operators are bounded. Thus one may take
$\Gamma_h^*=\Gamma_h$ for the induced norm used here.

\begin{theorem}[Practical bound at period boundaries]\label{thm:practical_boundary}
Let $t_k=T_o+2kh$. Under~\eqref{eq:residual_bound},
\begin{equation}
 \|z(t_{k+1})\|\leq\Gamma_h\bar\Delta_r,
 \qquad k\geq0.
 \label{eq:pdf_boundary_bound}
\end{equation}
\end{theorem}
\begin{proof}
Variation of constants over one period gives
\[
z(t_{k+1})=\Delta_Kz(t_k)+
\int_{t_k}^{t_{k+1}}\Psi(t_{k+1},s)B\Delta_r(s)\,ds.
\]
Theorem~\ref{thm:pdf_nullification} gives $\Delta_K=0$. Taking norms and
using~\eqref{eq:residual_bound} yields the result by the definition of
$\Gamma_h$.\end{proof}

\begin{corollary}[Uniform practical tracking bound]\label{cor:arbitrary_time}
There exists a finite constant $\Gamma_h^*$, depending only on the homogeneous
PDF dynamics and $h$, such that
\begin{equation}
 \|z(t)\|\leq\Gamma_h^*\bar\Delta_r,
 \qquad t\geq T_o+2h.
\end{equation}
The interval $[T_o,T_o+2h)$ contributes a finite transient determined by
$z(T_o)$; it is exactly nullified at $T_o+2h$ when $\Delta_r=0$.
\end{corollary}

If numerical implementation produces a one-period map satisfying
$\|\Delta_K\|\leq\varepsilon_\Delta<1$, then
\begin{equation}
 \|z_{k+1}\|\leq\varepsilon_\Delta\|z_k\|
 +\Gamma_h\bar\Delta_r,
 \qquad
 \limsup_{k\to\infty}\|z_k\|
 \leq\frac{\Gamma_h\bar\Delta_r}{1-\varepsilon_\Delta}.
 \label{eq:approx_nullification_bound}
\end{equation}
If $\|M_e^{-1}\|\leq\underline m^{-1}$ and
$\|\Delta d_o\|\leq\bar d_o$, $\|\Delta_m\|\leq\bar\Delta_m$,
and $\|\Delta_\tau\|\leq\bar\Delta_\tau$, then
\begin{equation}
 \bar\Delta_r\leq\underline m^{-1}
 (\bar d_o+\bar\Delta_m+\bar\Delta_\tau).
 \label{eq:explicit_delta_bar}
\end{equation}

\subsection{Task-space reference generation}
A damped generalized-Jacobian inverse may generate a joint reference by
\begin{equation}
 \dot q_d=J_g^\#[\dot x_e^d-K_x(x_e-x_e^d)],\qquad
 J_g^\#=J_g^\top(J_gJ_g^\top+\lambda^2I)^{-1}.
 \label{eq:task_space_velocity}
\end{equation}
Equation~\eqref{eq:task_space_velocity} is a reference-generation rule only.
Because damping and rank deficiency introduce a task-space residual, no exact
task-space convergence claim is made here; the theorem concerns joint-space
tracking.
